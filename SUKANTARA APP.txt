    kategoriSelect.addEventListener("change", renderTable);
    jantinaSelect.addEventListener("change", renderTable);
    rumahSelect.addEventListener("change", renderTable);
    acaraSelect.addEventListener("change", renderTable);

    function kategoriTahun(tahun) {
      if (tahun === "Tahun 1" || tahun === "Tahun 2") return "B8";
      if (tahun === "Tahun 3" || tahun === "Tahun 4") return "B10";
      if (tahun === "Tahun 5" || tahun === "Tahun 6") return "B12";
      return "";
    }

    function renderTable() {
      const autosync = true;
      const kategori = kategoriSelect.value;
      const jantina = jantinaSelect.value;
      const rumah = rumahSelect.value;
      const acara = acaraSelect.value;

      const acaraSah = {
        B8: ["60m", "100m"],
        B10: ["60m", "100m", "Lompat Jauh", "Lontar Peluru"],
        B12: ["100m", "200m", "Lompat Jauh", "Lontar Peluru"]
      };

      if (!acaraSah[kategori].includes(acara)) {
        jadualContainer.innerHTML = `<p class='warning'>Acara <strong>${acara}</strong> tidak ditawarkan untuk kategori <strong>${kategori}</strong>.</p>`;
        return;
      }

      const isPercubaan = acara === "Lompat Jauh" || acara === "Lontar Peluru";

      const baris = muridList.filter(([_, rumahMurid, tahun, jantinaMurid]) => {
        return jantinaMurid === jantina && rumahMurid === rumah && kategoriTahun(tahun) === kategori;
      });

      if (baris.length === 0) {
        jadualContainer.innerHTML = `<p class='warning'>Tiada murid dijumpai untuk pilihan ini.</p>`;
        return;
      }

      let html = `<table><thead><tr><th>Nama</th><th>Rumah</th><th>Kelas</th>`;
      html += isPercubaan ? `<th>Percubaan 1</th><th>Percubaan 2</th><th>Percubaan 3</th><th>Catatan Terbaik</th>` : `<th>Catatan</th>`;
      html += `<th>Tidak Hadir</th></tr></thead><tbody>`;

      baris.forEach(([nama, rumahMurid, tahun]) => {
        html += `<tr><td>${nama}</td><td>${rumahMurid}</td><td>${tahun}</td>`;
        if (isPercubaan) {
          html += `<td><input type='number' oninput='autoSave(this)'></td>`;
          html += `<td><input type='number' oninput='autoSave(this)'></td>`;
          html += `<td><input type='number' oninput='autoSave(this)'></td>`;
          html += `<td>-</td>`;
        } else {
          html += `<td><input type='text' oninput='autoSave(this)'></td>`;
        }
        html += `<td><input type='checkbox' oninput='autoSave(this)'></td></tr>`;
      });

      html += `</tbody></table>`;
      jadualContainer.innerHTML = html;
      loadSavedData();
    }

    function resetPilihan() {
      kategoriSelect.selectedIndex = 0;
      jantinaSelect.selectedIndex = 0;
      rumahSelect.selectedIndex = 0;
      acaraSelect.selectedIndex = 0;
      jadualContainer.innerHTML = "";
    }

    function autoSave(input) {
      const row = input.closest('tr');
      const cells = row.querySelectorAll('td');
      const nama = cells[0].textContent;
      const rumah = cells[1].textContent;
      const kelas = cells[2].textContent;
      const acara = document.getElementById('acara').value;
      const key = `${nama}-${acara}`;

      const inputs = row.querySelectorAll('input');
      const values = Array.from(inputs).map(i => i.type === 'checkbox' ? i.checked : i.value);
      localStorage.setItem(key, JSON.stringify(values));

      inputs.forEach(input => input.classList.add('saved'));
      setTimeout(() => inputs.forEach(input => input.classList.remove('saved')), 1000);

      const data = {
        nama,
        rumah,
        kelas,
        acara,
        catatan1: values[0] || "",
        catatan2: values[1] || "",
        catatan3: values[2] || "",
        terbaik: cells.length > 6 ? cells[6].textContent : "",
        tidakHadir: values[values.length - 1] || false
      };

      fetch("https://script.google.com/macros/s/AKfycbxWyvZsrNMMzhCoDVk7huQQdYHkVDyaCV_BXXLB6q2vuRzf18dnqSG-2C1pNiEaI-f3_w/exec", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(data)
      })
      .then(res => res.text())
      .then(response => console.log("Autosave ke Google Sheet:", response))
      .catch(error => console.error("Ralat autosave:", error));
    }

    function loadSavedData() {
      const rows = document.querySelectorAll('#jadualContainer tbody tr');
      rows.forEach(row => {
        const cells = row.querySelectorAll('td');
        const nama = cells[0].textContent;
        const acara = document.getElementById('acara').value;
        const key = `${nama}-${acara}`;
        const saved = JSON.parse(localStorage.getItem(key) || 'null');
        if (saved) {
          const inputs = row.querySelectorAll('input');
          saved.forEach((val, i) => {
            if (inputs[i].type === 'checkbox') {
              inputs[i].checked = val;
            } else {
              inputs[i].value = val;
            }
          });
        }
      });
    }

    function hantarKeGoogleSheet() {

          acara,

          catatan2,
          catatan3,

          tidakHadir
        };

        fetch("https://script.google.com/macros/s/AKfycbxWyvZsrNMMzhCoDVk7huQQdYHkVDyaCV_BXXLB6q2vuRzf18dnqSG-2C1pNiEaI-f3_w/exec", {

      });
    }
